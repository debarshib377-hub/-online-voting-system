const express = require("express"); const bodyParser = require("body-parser"); const bcrypt = require("bcrypt"); const app = express(); app.use(bodyParser.json()); let users = []; let votes = []; app.post("/register", async (req, res) => { const { email, password } = req.body; const hashed = await bcrypt.hash(password, 10); users.push({ email, password: hashed, voted: false }); res.send("User registered"); }); app.post("/login", async (req, res) => { const { email, password } = req.body; const user = users.find(u => u.email === email); if (!user) return res.send("User not found"); const match = await bcrypt.compare(password, user.password); if (!match) return res.send("Wrong password"); res.send("Login successful"); }); app.post("/vote", (req, res) => { const { email, candidate } = req.body; const user = users.find(u => u.email === email); if (user.voted) return res.send("Already voted"); votes.push(candidate); user.voted = true; res.send("Vote cast successfully"); }); app.listen(3000, () => console.log("Server running")); body { font-family: Arial; text-align: center; } form { margin: 20px auto; width: 300px; } input, button { display: block; margin: 10px auto; padding: 10px; }
